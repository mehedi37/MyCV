\section{Thesis and Research}

\datedsubsection{TRACE: Curriculum Bottleneck Detection and Causal Validation}{Jul 2025 -- Aug 2026}
\textit{B.Sc.\ Thesis, Dept.\ of CSE, RUET}
\begin{itemize}
    \resitem{Started from my own class, where about a third failed one first-year mathematics course (about 4\% fail a typical course). Does such a course hold students back, or only show who was already behind?}
    \resitem{Built TRACE, which ranks courses by prerequisite-graph position (NetworkX, GAT), a Bayesian fail-rate model, and student-feedback sentiment (DeBERTa, VADER).}
    \resitem{Reached F-measure 0.75 and NDCG@10 0.81 on a synthetic curriculum with planted bottlenecks; on a local university's curriculum, a domain expert agreed with 8 of its top 10 courses.}
    \resitem{Used causal inference to test what drives failure in a flagged module: on 23,369 OULAD students, DoWhy estimated that high assignment engagement raised held-out pass rates by 19.9 points ($p<0.01$).}
\end{itemize}
