\datedsubsection{TRACE: A Multi-Modal Framework for Curriculum Bottleneck Detection and Causal Validation}{Jul 2025 -- Aug 2026}
\textit{B.Sc. Thesis, Dept. of CSE, RUET} | \textit{Supervisor:} Farjana Parvin, Assistant Professor
\begin{itemize}
    \resitem{\textbf{Architected} TRACE, a multi-modal graph-mining framework fusing prerequisite-graph centrality (NetworkX, GAT edge weighting), Bayesian fail-rate modeling, and aspect-based sentiment analysis (DeBERTa/VADER) into a single composite bottleneck score.}
    \resitem{\textbf{Built} a reproducible synthetic-curriculum benchmark with planted, type-disjoint bottlenecks; fusion reached an F-measure of 0.75 and NDCG@10 of 0.81, roughly double the best single-signal baseline.}
    \resitem{\textbf{Causally validated} flagged bottlenecks on OULAD (23,369 students) via \textbf{DoWhy} backdoor-adjustment OLS; high assignment engagement raised held-out pass rates by 19.9 percentage points ($p<0.01$, E-value 1.89).}
    \resitem{\textbf{Demonstrated} that the structure-only detector generalizes across RUET, MIT, and Stanford curricula (top-5 hub courses gate 44--55\% of the downstream curriculum) and validated rankings against a domain expert (8/10 agreement); surfaced RUET's high-failure MATH1213 (16.9\% fail rate) via the ``Preparation Paradox,'' where student effort masks true course difficulty.}
\end{itemize}
