\datedsubsection{PharmaC \textbar\ Desktop Pharmacy Management System \href{https://www.mehedi-hasan-maruf.me/projects/pharmac}{\small\faExternalLink*}}{Feb 2025}
\begin{itemize}
    \resitem{Built a WPF desktop app (.NET 8, MVVM, WPF-UI) for pharmacy inventory, sales, and orders.}
    \resitem{Modeled medicines, brands, generics, and admin accounts in Entity Framework Core (SQL Server, MySQL).}
    \resitem{Implemented JWT authentication with token refresh, barcode lookup for medicine scanning, and per-device licensing from a CPU/motherboard/BIOS hardware fingerprint.}
    \resitem{Connected the app to a REST API (\texttt{HttpClient}) for medicines, orders, carts, and licenses, so inventory syncs in real time and checkout runs in BDT.}
\end{itemize}
