\datedsubsection{Omnious \textbar\ Visual Debugging Platform \href{https://www.mehedi-hasan-maruf.me/projects/omnious}{\small\faExternalLink*}}{Nov 2025 -- Present}
\begin{itemize}
    \resitem{Building a debugger that draws a program's code graph, animates execution traces, shows where errors cluster, and lets users ask an AI assistant about them.}
    \resitem{A TypeScript Turborepo monorepo: Fastify 5 and tRPC v11 backend, Next.js 16 / React 19 frontend.}
    \resitem{Stores data in PostgreSQL 15 on Supabase, using pgvector, ltree, and pg\_trgm for semantic search and tree-shaped data, with row-level security.}
    \resitem{Runs in Docker Compose with multi-stage builds and deploys through GitHub Actions.}
\end{itemize}
